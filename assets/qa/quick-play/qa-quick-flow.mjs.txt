async function startServer(extra) {
  server = spawn("node", ["apps/server/src/main.ts"], { cwd: REPO, env: { ...process.env, DOMINO_DATA_DIR: data, PORT: "8787", HOST: "127.0.0.1", DOMINO_ALLOWED_ORIGINS: "http://localhost:5174", DOMINO_QUICK_PLAY_FILL_SECONDS: "8", ...extra }, stdio: ["ignore", "inherit", "inherit"] });
  for (let i = 0; i < 50; i++) { try { if ((await fetch(`${API}/health`)).ok) return; } catch {} await sleep(100); }
  throw new Error("servidor no arrancó");
}
async function stopServer() { server.kill("SIGTERM"); await new Promise((resolve) => server.once("exit", resolve)); }

"GET" : "POST", headers: { "content-type": "application/json", ...(token ? { authorization: `Bearer ${token}` } : {}) }, ...(body === undefined ? {} : { body: JSON.stringify(body) }) }); return response.json(); };
